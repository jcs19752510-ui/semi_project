# 1. 현재 브랜치 위치 확인
git branch

# 2. 부모 없는 새 브랜치 생성 (히스토리 없음, 현재 파일은 그대로 유지)
git checkout --orphan temp_branch

# 3. 모든 파일 스테이징 (현재 작업 디렉토리 기준)
git add -A

# 4. 새 커밋 생성 (이게 유일한 첫 커밋이 됨)
git commit -m "히스토리 초기화 - 대용량 파일 제거"

# 5. 기존 브랜치(SEMI_PROJECT) 삭제하고 temp_branch를 그 이름으로 변경
git branch -D SEMI_PROJECT
git branch -m SEMI_PROJECT

# 6. 로컬 가비지 컬렉션으로 기존 오브젝트 정리
git gc --aggressive --prune=now

# 7. 원격에 강제 푸시 (기존 원격 히스토리 완전히 대체)
git push origin SEMI_PROJECT --force
